class Solution {
    public int solution(int n) {
        int answer = 0;
        int tn = n;
        int cnt = 1; //1의 개수
        while(tn > 1){
            int t = tn % 2;
            if(t == 1){
                cnt++;
            }
            tn = tn / 2;
        }
        
        for(int i = n+1; i<=1000000; i++){
            int cntt = 1; 
            int ttt = i;
            
            while(ttt > 1){
                int t = ttt % 2;
                if(t == 1){
                    cntt++;
                }
                ttt =ttt / 2;
            }
            
            if(cnt == cntt){
                answer = i;
                break;
            }
        }
     
        return answer;
    }
}

class Solution {
    public int solution(int n) {
        int answer = 0;
        int tn = n;
        int cnt = 0;
        cnt = countOne(tn);
        
        for(int i = n+1; i<=1000000; i++){
            int cntt = 1; 
            int ttt = i;
           cntt = countOne(ttt);
        if(cnt == cntt){
            answer = i;
            break;
        }
        }
     
        return answer;
    }
    
    
    public int countOne(int num){
        int cnt = 1;
          while(num > 1){
            int t = num % 2;
            if(t == 1){
                cnt++;
            }
            num = num / 2;
        }
        return cnt;
    }
}
